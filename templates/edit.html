// 🌟 即時組字串流：只要按鍵盤敲注音，0 毫秒即刻通知大螢幕與搭檔 let lastSentComposing = ''; function broadcastLiveTyping(str) { if (str !== lastSentComposing) { lastSentComposing = str; // 1. 廣播給搭檔的遠端游標 socket.emit('cursor_move', { room: roomId, cursor_index: editor.selectionStart, typing_text: str }); // 2. 廣播給觀眾端大螢幕即時浮現字詞 socket.emit('live_composing', { room: roomId, composing_text: str }); } } function bindEditorEvents() { editor.addEventListener('compositionstart', (e) => { isComposing = true; broadcastLiveTyping(e.data || '...'); }); // 監聽微軟注音敲擊跳動 editor.addEventListener('compositionupdate', (e) => { broadcastLiveTyping(e.data || ''); }); editor.addEventListener('input', (e) => { // 若在注音組字中，即時撈出字元廣播給大螢幕 if (e.isComposing || isComposing) { if (e.data) broadcastLiveTyping(e.data); return; } handleLocalTyping(); }); // 按 Enter 確定定稿 editor.addEventListener('compositionend', () => { isComposing = false; broadcastLiveTyping(''); // 清空大螢幕幽靈字 handleLocalTyping(); // 固化正式字元 }); editor.addEventListener('click', broadcastCursorOnly); editor.addEventListener('keyup', (e) => { if (['ArrowLeft','ArrowRight','ArrowUp','ArrowDown','Home','End','Enter'].includes(e.key)) { broadcastCursorOnly(); } }); editor.addEventListener('scroll', () => { redrawAllRemoteCursors(); updateEndLinePosition(); }); } bindEditorEvents();
